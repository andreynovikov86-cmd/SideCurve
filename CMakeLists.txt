cmake_minimum_required(VERSION 3.25)
project(SideKick VERSION 0.5.5 LANGUAGES C CXX)
if(APPLE)
 enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(VST3_SDK_ROOT "" CACHE PATH "Official Steinberg VST3 SDK with submodules")
if(NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
 message(FATAL_ERROR "Set VST3_SDK_ROOT to the official SDK checkout")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT ON CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(VSTGUI_STANDALONE OFF CACHE BOOL "" FORCE)
set(VSTGUI_ENABLE_WAYLAND_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk-build)
set(SDK_ROOT "${VST3_SDK_ROOT}")
set(public_sdk_SOURCE_DIR "${VST3_SDK_ROOT}/public.sdk")
smtg_add_vst3plugin(SideKick PACKAGE_NAME SideCurve SOURCES_LIST src/Plugin.cpp src/Plugin.h src/Editor.cpp src/Editor.h src/Engine.h src/EditorModel.h src/Branding.h src/Preferences.h)
target_link_libraries(SideKick PRIVATE sdk vstgui_support)
target_compile_definitions(SideKick PRIVATE SMTG_USE_STDATOMIC_H=0)
if(APPLE)
 smtg_target_set_bundle(SideKick BUNDLE_IDENTIFIER "com.andreynovikov.sidekick" COMPANY_NAME "Andrey Novikov")
 set_target_properties(SideKick PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in" XCODE_ATTRIBUTE_INFOPLIST_FILE "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in")
 add_custom_command(TARGET SideKick POST_BUILD COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:SideKick>")
endif()
add_executable(sidekick_trace tests/trace.cpp)
target_include_directories(sidekick_trace PRIVATE src)
enable_testing()
add_executable(sidekick_tests tests/engine_tests.cpp)
target_include_directories(sidekick_tests PRIVATE src)
add_test(NAME engine COMMAND sidekick_tests)
